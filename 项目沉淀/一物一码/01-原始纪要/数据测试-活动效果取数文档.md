    COUNT(DISTINCT h.member_id)   AS member_cnt,   
    COUNT(DISTINCT h.wx_unionid)  AS unionid_cnt

    lucky_wecom.t_wecom_external_user_history h
INNER JOIN
    lucky_wecom.t_wecom_external_user u
    ON h.external_user_id = u.external_user_id

    h.state = 'rtd瓶装饮料-瓶内营销码现金红包'（待提供真实名称）

SELECT
    COUNT(DISTINCT r.member_id) AS expand_prize_user_num
FROM
    lucky_epromotion.t_rtd_cap_activity_record_user_line r
INNER JOIN
    lucky_epromotion.t_rtd_cap_activity_prize p
    ON r.prize_row_id = p.id
WHERE
    r.create_time >= '2026-06-20 00:00:00'
    AND r.create_time < '2026-07-06 00:00:00'
    AND r.record_prize_status = 3   -- 3=膨胀路径
    AND p.is_expand = 1             -- 该奖品行支持膨胀

    COUNT(DISTINCT member_id)          AS paid_user_num,          -- 领到红包人数

    lucky_epromotion.t_rtd_redpack_user_instance
WHERE
    transfer_done_time >= '2026-06-20 00:00:00'
    AND transfer_done_time < '2026-07-06 00:00:00'
